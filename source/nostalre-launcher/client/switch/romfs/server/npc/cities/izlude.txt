-	script	::Sailor_izlude	-1,{
	mes "[Marinheiro]";
	mes "Olá pessoal!";
	mes "Atenção, atenção!";
	mes "Venha e monte o vento";
	mes "em um navio fascinante!";
	mes "Depressa, depressa!";
	next;
	switch(select("Ilha Byalan -> 150 Zeny.:Marina de Alberta -> 500 Zeny.:Cancelar.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Marinheiro]";
			mes "150 Zeny!";
			mes "Apenas 150 Zeny para andar!";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Marinheiro]";
			mes "500 Zeny!";
			mes "Apenas 500 Zeny para andar!";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
