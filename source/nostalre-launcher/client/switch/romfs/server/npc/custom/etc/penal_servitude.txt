sec_pri,36,58,1	script	Chief Warder	105,{ 
	mes "[Saddeus]";
	emotion ET_QUESTION;
	mes (Sex==SEX_MALE?"Sr.":"Sra.")+"Prisioneiro nº"+BaseLevel+JobLevel+",";
	mes "que barulho é esse aí?";
	next;
	menu "Me deixe sair, eu pago!",-, "Deixe-me trabalhar para conquistar minha liberdade!",M_Q1, "Nada",M_NO_THANKS;

M_PAY:
	mes "[Saddeus]";
	set .@MUSTPAY,(PRISON+1)*1000000;
	if (.@MUSTPAY<1000000) set .@MUSTPAY,1000000;
	if (.@MUSTPAY>100000000) set .@MUSTPAY,100000000;
	if (PRISON==0) mes "Não consigo me lembrar de você. É a sua primeira vez aqui?";
	if (PRISON>0) mes "Você de novo? Alguém não aprendeu nada desde a primeira visita...";
	mes "Tudo bem, poderíamos libertar você por ^FF0000"+.@MUSTPAY+"z^000000.";
	next;
	menu "Eu pago em dinheiro!",M_PAYCASH,"Vocês aceitam cheque?",M_PAYBANK,"Mudei de ideia.",-;

	mes "[Saddeus]";
	mes "Você tem um tempo para pensar sobre isso...";
	close;

M_PAYCASH:
	if (.@MUSTPAY>Zeny) goto L_NOCASH;
	set Zeny,Zeny-.@MUSTPAY;
	mes "[Saddeus]";
	mes "OK, assine aqui e ali.";
	goto L_RELEASE;

L_NOCASH:
	mes "[Saddeus]";
	mes "O que é isso? Não é o suficiente!";
	close;

M_PAYBANK:
	if (.@MUSTPAY>#kafrabank) goto L_NOBANK;
	set #kafrabank,#kafrabank-.@MUSTPAY;
	mes "[Saddeus]";
	mes "OK, assine seu cheque. E coloque seu nome no meu livro.";
	goto L_RELEASE;

L_NOBANK:
	mes "[Saddeus]";
	if (#kafrabank==0) mes "Seu processo não diz nada sobre suas contas bancárias.";
	if (#kafrabank!=0) mes "Infelizmente, aqui estão as informações da sua conta... ^FF0000"+#kafrabank+"z^000000... no processo. Não é o suficiente.";
	mes "Pare com esses joguinhos agora!";
	close;

L_RELEASE:
	set PRISON,PRISON+1;
	next;
	mes "[Saddeus]";
	mes "Você está livre agora!";
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

M_Q1:
	mes "[Saddeus]";
	mes "Nosso estoque de cogumelos está baixo. Lembra daquela sua comida de manhã? Que tal trabalhar em nossa fazenda de cogumelos patrocinada?";
	next;
	menu "OK, eu vou trabalhar para sair!",-, "Eu amo esta prisão!",M_NO_THANKS;

	mes "[Saddeus]";
	if(checkcart() || checkfalcon() || checkriding()){
		emotion ET_SCRATCH;
		mes "Sinto muito, você deve liberar seu Carrinho, PecoPeco ou Falcão!";
		close;
	}
	mes "Fale com nosso supervisor, Oliver.";
	next;

	nude;

	set .@rw,rand(1,4);
	if (.@rw==2) goto L_W2;
	if (.@rw==3) goto L_W3;
	if (.@rw==4) goto L_W4;

L_W1:
	savepoint "sec_in02",179,76;
	warp "sec_in02",179,76;
	close;

L_W2:
	savepoint "sec_in02",139,32;
	warp "sec_in02",139,32;
	close;

L_W3:
	savepoint "sec_in02",100,28;
	warp "sec_in02",100,28;
	close;

L_W4:
	savepoint "sec_in02",107,75;
	warp "sec_in02",107,75;
	close;

M_NO_THANKS:
	mes "[Saddeus]";  
	if (rand(2)) mes "Hoje é Natal, é?";
	mes "Agora cale a boca e dê o fora!";
	if (rand(2)) emotion ET_HUK;
	close;  
}

sec_in02,137,57,1	script	Overseer	708,{ 
	mes "[Oliver]";

	delitem 4002,countitem(4002);//Items: Fabre_Card,
	delitem 4009,countitem(4009);//Items: Chonchon_Card,
	delitem 4022,countitem(4022);//Items: Spore_Card,
	delitem 4048,countitem(4048);//Items: Poison_Spore_Card,
	
	if (PRISON_Q <= 0 ) goto L_GET_Q;

	mes "Eu quero que você reúna ^FF0000"+PRISON_Q+"Cogumelo de Rede Laranja^000000.";
	if ( countitem(1069)<PRISON_Q ) close;//Items: Orange_Net_Mushroom,
	mes "OK... me entregue todos os cogumelos...";
	mes "Deixe-me ver..."+countitem(1069)+"de quase comestíveis...";//Items: Orange_Net_Mushroom,
	mes "E"+countitem(1070)+"os inúteis mortais.";//Items: Orange_Gooey_Mushroom_,
	next;
	mes "[Oliver]";
	mes "Obrigado. Você está livre!";
	set PRISON_Q,0;
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,
	next;
	savepoint "izlude",105,112;
	warp "izlude",105,112;
	close;

L_GET_Q:
	set PRISON,PRISON+1;
	if (PRISON > 1) mes "Hmm... Você já esteve aqui antes... Ora, ora...";

	if ( countitem(1069) || countitem(1070) ) mes "Jogue esses cogumelos fora!! Eles são venenosos!";//Items: Orange_Net_Mushroom, Orange_Gooey_Mushroom_,
	delitem 1069,countitem(1069);//Items: Orange_Net_Mushroom,
	delitem 1070,countitem(1070);//Items: Orange_Gooey_Mushroom_,

	set PRISON_Q, PRISON*3 + BaseLevel/3;
	set PRISON_Q, PRISON_Q * (readparam(bAspd)/55 + 1);
	set PRISON_Q, PRISON_Q+rand(50,60);
	mes "OK, você deve me trazer ^FF0000"+PRISON_Q+"^000000 comestíveis ^FF0000Cogumelo de Rede Laranja^000000 cogumel.. cogumelos.";
	if(readparam(bAspd)>=100) mes "Você é rápido! Então você terá que conseguir mais cogumelos.";
	if (rand(10)<4) mes "E você pode comer os mortais no café da manhã.";
	close;
}
