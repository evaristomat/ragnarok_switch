valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Sniper]";
			mes "Parabéns.";
			mes "Honra aos guerreiros!";
			close;
		}
		mes "[Sniper]";
		mes "Um tiro.";
		mes "Uma morte.";
		mes "Não é tão difícil";
		mes "quando você desenvolve a";
		mes "visão para esse estilo";
		mes "de batalha.";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Sniper]";
		mes "O mundo está";
		mes "precisando de arqueiros poderosos";
		mes "como você. Você está pronto para";
		mes "essa incrível responsabilidade?";
		next;
		mes "[Sniper]";
		mes "Você está disposto a";
		mes "dar o próximo passo e";
		mes "se tornar um Atirador de Elite?";
		next;
		if (select("Não.:Sim.") == 1) {
			mes "[Sniper]";
			mes "Quando estiver pronto,";
			mes "sinta-se à vontade para voltar.";
			mes "Honra aos guerreiros!";
			close;
		}
		if (SkillPoint) {
			mes "[Sniper]";
			mes "Ainda é possível que você aprenda mais habilidades. Por favor, use";
			mes "todos os seus pontos de habilidade restantes antes de voltar para mim.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Sniper]";
		mes "Parabéns!";
		mes "Como um Atirador de Elite, espero";
		mes "que os servos do mal";
		mes "nunca estejam seguros enquanto";
		mes "estiverem no seu";
		mes "campo de visão!";
		close;
	}
	mes "[Sniper]";
	mes "Bem-vindo";
	mes "a Valhalla,";
	mes "o Salão da Honra.";
	next;
	mes "[Sniper]";
	mes "Por favor, fique";
	mes "à vontade";
	mes "enquanto estiver aqui.";
	mes "Honra aos guerreiros!";
	close;
}
