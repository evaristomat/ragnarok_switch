//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[Equipe da Cool Event Corp.]";
	mes "Bem-vindo à Cool Event Corp.";
	mes "Nossa equipe está sempre trabalhando";
	mes "superar suas expectativas";
	mes "para um serviço de qualidade. Então, como";
	mes "posso ajudá-lo hoje?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[Equipe da Cool Event Corp.]";
		mes "Seu ponto de respawn";
		mes "foi salvo aqui";
		mes getarg(1)+".";
		mes "Obrigado por usar o";
		mes "Serviço legal da Event Corp.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[Equipe da Cool Event Corp.]";
		if(!callfunc("F_CanOpenStorage")){
			mes "Me desculpe, mas você";
			mes "preciso do novato";
			mes "Nível de habilidade básica 6 a";
			mes "use o serviço de armazenamento.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Me desculpe, mas você não";
			mes "tenha Zeny suficiente para usar";
			mes "o Serviço de Armazenamento. Nosso";
			mes "A taxa de acesso ao armazenamento é 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Deixe-me abrir seu pessoal";
		mes "armazenamento para você imediatamente.";
		mes "Obrigado por apoiar legal";
		mes "Event Corp. usando nosso";
		mes "serviços. Tenha um bom dia ~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Por favor escolha";
		mes "seu destino.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:Cancelar") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[Equipe da Cool Event Corp.]";
				mes "Sinto muito, mas você não tem";
				mes "zeny suficiente para o teleporte";
				mes "Serviço. A taxa para se teletransportar";
				mes "to "+ .@destination$ +" is "+.@cost+" Zeny.";
			}
		}
		close2;
		break;
	case 4:
		mes "[Equipe da Cool Event Corp.]";
		if(BaseClass != Job_Merchant){
			mes "Sinto muito, mas o";
			mes "Serviço de aluguel de carrinhos";
			if (checkre(0))
				mes "está disponível apenas para classes Merchant.";
			else {
				mes "está disponível apenas para comerciantes,";
				mes "Ferreiros, Ferreiros Brancos,";
				mes "Alquimistas e Criadores.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Você só pode alugar um carrinho depois";
			mes "aprendendo a habilidade Pushcart.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Você já tem";
			mes "um carrinho de mão equipado.";
			mes "Infelizmente, não podemos";
			mes "alugue mais de um para";
			mes "cada cliente por vez.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "O aluguel do carrinho";
		mes "a taxa é 800 Zeny. Iria";
		mes "você gosta de alugar um Pushcart?";
		next;
		if(select("Alugue um carrinho de mão:Cancelar") == 1){
			if(Zeny < 800){
				mes "[Equipe da Cool Event Corp.]";
				mes "Me desculpe, mas você";
				mes "não tenho o suficiente";
				mes "Zeny vai pagar o carrinho";
				mes "taxa de aluguel de 800 Zeny.";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[Equipe da Cool Event Corp.]";
		mes "Cool Event Corp. está sempre";
		mes "esforçando-se para oferecer o melhor";
		mes "serviços para nossos clientes.";
		mes "Ajude-nos a ser os melhores";
		mes "nos fornecendo suas opiniões";
		mes "e feedback honesto. Obrigado.";
		close2;
	}
	cutin "", 255;
	end;
}

		"Save:Use Storage::Rent a Pushcart:Storage Password Service:Cancel",
		"in the town of Lighthalzen","lighthalzen",95,240;
	end;
}
