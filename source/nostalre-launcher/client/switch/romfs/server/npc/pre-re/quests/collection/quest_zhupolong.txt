	if (checkquest(60182) != -1) {
		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[Dragon Hunter]";
			mes "Você terminou de caçar os 50 Zhu Po Long?";
			next;
			switch(select("Sim:Não:Quero desistir")) {
			case 1:
				mes "[Dragon Hunter]";
				mes "Hmm, acho que você ainda não caçou 50 deles.";
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Lembre-se, preciso de ajuda para caçar Zhu Po Long, então vá e cace 50 deles.";
				close;
			case 3:
				mes "[Dragon Hunter]";
				mes "Tem certeza de que deseja parar a caçada?";
				mes "Qualquer progresso que você tenha feito será apagado.";
				next;
				switch(select("Sim:Não")) {
				case 1:
					mes "[Dragon Hunter]";
					mes "Tudo bem então, volte aqui se mudar de ideia.";
					erasequest 60182;
					close;
				case 2:
					mes "[Dragon Hunter]";
					mes "Por favor, mate 50 Zhu Po Long para tornar este um lugar onde as pessoas possam vir e ganhar Experiência.";
					close;
				}
			}
		}
		mes "[Dragon Hunter]";
		mes "Incrível, você fez isso rapidamente.";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[Dragon Hunter]";
		mes "Oh, ótimo!";
		mes "Você está aqui para me ajudar, certo?";
		mes "Se não, não sei o que fazer com os monstros neste lugar!";
		next;
		switch(select("Com certeza:Não")) {
		case 1:
			mes "[Dragon Hunter]";
			mes "Ótimo! Eu sabia, no momento em que te vi, que poderia contar com você!";
			mes "Preciso que você reduza a quantidade de monstros que estão causando tumulto neste mapa!";
			mes "Há muitos Zhu Po Long por perto, o que torna impossível para os jogadores de nível baixo ganharem uma boa experiência aqui.";
			next;
			mes "[Dragon Hunter]";
			mes "Então, você ainda gostaria de me ajudar?";
			next;
			switch(select("Com certeza:Cancelar")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[Dragon Hunter]";
					mes "Obrigado pela sua grande ajuda com os Zhu Po Long!";
					mes "Atualmente a situação está estável e não preciso mais da sua ajuda para matá-los.";
					close;
				}
				mes "[Dragon Hunter]";
				mes "Mate 50 desses Zhu Po Long e me avise quando terminar. Eu recompensarei seus esforços.";
				setquest 60182;
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Se você mudar de ideia, por favor, volte.";
				close;
			}
		case 2:
			mes "[Dragon Hunter]";
			mes "Se você mudar de ideia, por favor, volte.";
			close;
		}
	}
	mes "[Dragon Hunter]";
	mes "Oh céus, oh céus!";
	next;
	mes "[Dragon Hunter]";
	mes "Desculpe, não posso falar agora, estou morrendo de dor aqui, esses malditos Zhu Po Long vão me matar.";
	close;
}
