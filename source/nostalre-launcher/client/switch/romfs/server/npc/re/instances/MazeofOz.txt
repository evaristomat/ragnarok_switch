			switch( instance_enter(.@md_name$) ) {
			case IE_OTHER:
				mes "[Disguised Miriam]";
				mes "^ff0000Ocorreu um erro desconhecido.^000000";
				close;
			case IE_NOINSTANCE:

				mes "Vamos esperar um pouco e tentar novamente.";
				close;
			case IE_NOMEMBER:
				mes "[Disguised Miriam]";
				mes "^ff0000Apenas um membro do grupo pode entrar na Instância.^000000";
				close;
			case IE_OK:

			end;
