pay_arche,44,124,5	script	Marry Jae	4_F_JOB_BLACKSMITH,{
	mes "[Marry Jae]";
	if (BaseLevel < 60) {
		mes "Seu nível base está muito baixo, volte quando atingir o nível 60+.";
		close;
	}
	mes "^ff0000Se você tentar domar algum monstro, isso não contará como uma eliminação. Por favor, tome cuidado.^000000";
	next;
	.@day = gettime(DT_DAYOFWEEK);
	if (.@day != SATURDAY && .@day != SUNDAY) {
		mes "[Marry Jae]";
		mes "Oh, um curioso observador? Eu sou Marry Jae, a pessoa responsável pelos finais de semana entre os dias da semana.";
		next;
		mes "[Marry Jae]";
		mes "O que é a masmorra de fim de semana? Na verdade, é um espaço projetado especialmente para novos aventureiros treinarem.";
		next;
		mes "^0000FFOlhando ao redor, é óbvio que Marry Jae está falando com você.^000000";
		next;
		mes "[Marry Jae]";
		mes "Para aqueles que não têm um grupo nos fins de semana, este lugar é um campo de caça e ganho de experiência para aventureiros solitários!";
		next;
		mes "[Marry Jae]";
		mes "Além disso, o ganho de experiência aqui é muito generoso.";
		next;
		select( "Quando é o horário de funcionamento?" );
		mes "[Marry Jae]";
		mes "A masmorra ficará aberta durante:";
		mes "^0000FFSábado 00:00 ~ Domingo 23:59^000000";
		next;
		mes "[Marry Jae]";
		mes "Espero ver você quando o fim de semana chegar~";
		close;
	}
	.@md_name$ = "Weekend Dungeon";

	if (getcharid(1) < 1) {
		mes "[Marry Jae]";
		mes "Você precisa formar um grupo de um ou mais membros.";
		close;
	}
	switch( checkquest(12378,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		switch( checkquest(12377,PLAYTIME) ) {
		case -1:
		case 2:
			mes "[Marry Jae]";
			mes "Desculpe, mas o período de entrada para a" + .@md_name$ + "expirou.";
			mes "Descanse enquanto espera a masmorra ficar disponível novamente.";
			close;
		case 0:
		case 1:
			break;
		}
		break;
	case 2:
		erasequest 12377;
		erasequest 12378;
		mes "[Marry Jae]";
		mes "O tempo de espera expirou.";
		mes "Você pode entrar novamente na Masmorra Memorial.";
		close;
	}

	if (is_party_leader() == true)
		.@menu$ = "Preparar Masmorra Memorial";

	mes "[Marry Jae]";
	mes "Acho que você está quase pronto.";
	mes "Agora, você gostaria de entrar nela agora?";
	next;
	switch( select( .@menu$, "Entrar na Masmorra de Fim de Semana", "Cancelar" ) ) {
	case 1:
		instance_create(.@md_name$);
		mes "[Marry Jae]";
		mes "A Masmorra Memorial foi criada.";
		mes "Você já pode entrar.";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			end;
		case IE_NOINSTANCE:
			mes "Não há nenhuma Masmorra Memorial registrada.";
			close;
		case IE_OTHER:
			mes "Ocorreu um erro desconhecido.";
			close;
		case IE_OK:
			mes "[Marry Jae]";
			mes "Pode ir!";
			mapannounce "pay_arche", "" + strcharinfo(0) + "do grupo," + getpartyname( getcharid(1) ) + ", is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12377) == 0)
				setquest 12377;
			if (isbegin_quest(12378) == 0)
				setquest 12378;
			end;
		}
		end;
	case 3:
		mes "[Marry Jae]";
		mes "Você não se preparou?";
		mes "Acho que não tem jeito.";
		close;
	}
}

		close;
	}
	if (checkweight(23144,1) == 0) {
		mes "Você tem muitos tipos de objetos. Esvazie seu inventário.";
		close;

		mes "Não posso continuar porque você tem muitos objetos pesados. Reduza seu peso total esvaziando seu inventário.";
		close;
	}

	next;
	if (select( "Sim", "Não" ) == 2) {
		mes "[Gift Supplies Clerk]";
		mes "Volte sempre que precisar de serviços de embalagem!";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
}
