prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Father Mareusis]";
				mes "Ah, sinto que você suportou";
				mes "uma experiência de vida passada. Você deve ter aprendido muitas coisas antes de entrar em Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Father Mareusis]";
					mes "Infelizmente, não creio que você esteja pronto para se tornar um Noviço ainda. Por favor, termine de aprender todas as Habilidades Básicas primeiro.";
					next;
					mes "[Father Mareusis]";
					mes "Enquanto isso,";
					mes "esperarei até que";
					mes "você esteja pronto.";
					mes "Que Deus esteja";
					mes "com você.";
					close;
				}
				mes "[Father Mareusis]";
				mes "Bem, dou-lhe as boas-vindas";
				mes "de volta de Valhalla e";
				mes "desejo-lhe sorte em sua";
				mes "nova jornada de vida.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Father Mareusis]";
				mes "Agora, aventure-se e busque aqueles que precisam de sua ajuda. Que Deus ilumine seu caminho.";
				close;
			}else{
				mes "[Father Mareusis]";
				mes "Agora, aventure-se para buscar pessoas que precisam de sua ajuda. Que Deus ilumine o seu caminhar.";
				close;
			}
		}else{
			mes "[Father Mareusis]";
			mes "Sinto que você suportou uma experiência de vida passada. Você deve ter aprendido muitas coisas antes de entrar em Valhalla.";
			next;
			mes "[Father Mareusis]";
			mes "No entanto, posso dizer que você não é adequado para ser um Noviço. Por favor, lembre-se de quem você foi em sua vida passada e encontre seu caminho.";
			close;
		}
	}
	mes "[Father Mareusis]";
	mes "O que é que você procura?";
	next;
	switch(select("Mudar sua classe para Noviço.:Perguntar os requisitos para ser um Noviço.:Sair.")) {
	case 1:
		mes "[Father Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Você está se sentindo bem hoje? Posso dizer pelo seu traje que você já é um Noviço. Você não está brincando, está?";
			else
				mes "Sinto muito, mas parece que você já tem sua própria classe, não é?";
			close;
		}else{
			mes "Você deseja verdadeiramente se tornar um servo de Deus?";
			mes "Vamos ver se você está pronto para isso ou não... Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Father Mareusis]";
				mes "Ora, puxa?! Você ainda não completou o treinamento básico?! Você tem um longo caminho pela frente! Volte após aumentar seu nível de classe!";
				close;
			}
			mes "[Father Mareusis]";
			mes "Hmm... seu nível de classe é suficiente...";
			mes "Bom. Agora lhe darei a qualificação para se tornar um Noviço.";
			next;
			mes "[Father Mareusis]";
			mes "Lembre-se sempre de ser grato a Deus, que cuida de nós o tempo todo. No caos e em tempos de dificuldade, enfrente suas provações com fé inabalável.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Father Mareusis]";
			mes "Por último, quero sinceramente parabenizá-lo por perseverar em sua provação de penitência.";
			close;
		}
	case 2:
		mes "[Father Mareusis]";
		mes "Você deseja se tornar um Noviço?";
		mes "Então, você deve cumprir os seguintes requisitos, pensando neles como as práticas dadas por Deus.";
		next;
		mes "[Father Mareusis]";
		mes "Primeiro, você deve alcançar pelo menos o Nível de Classe 9 de Aprendiz e aprender todas as Habilidades Básicas.";
		mes "Esta é a coisa mais básica a se fazer, então você precisa encará-la como uma forma de treinamento pessoal.";
		next;
		mes "[Father Mareusis]";
		mes "Quando achar que cumpriu este requisito, volte a mim novamente. Então você terá uma classe sagrada na qual poderá espalhar a vontade de Deus.";
		close;
	case 3:
		close;
	}
}
