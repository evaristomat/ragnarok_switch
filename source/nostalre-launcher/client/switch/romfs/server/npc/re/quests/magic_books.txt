		close;

	.@req_amount = getarg(2);

		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "Você usou"+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +"para criar ^999900"+ getarg(3) +"Essência^000000.";
			next;
			mes "^008800A"+getarg(3)+"Essência durará permanentemente, a menos que haja algum problema.^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888Você falhou ao tentar refinar a essência usando"+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +".^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "Você já criou"+getarg(3)+"Essência nesta área.";
		close;
	}
	mes "Alguém já usou esta área.";
	close;
}
