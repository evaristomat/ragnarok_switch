moc_prydb1,103,54,3	script	Suspicious Cat#night	547,{
	mes "[Suspicious Cat]";
	mes "Não acredito que uma múmia pôde fazer isso comigo. Ah, cara, dói muito. Ninguém pode saber disso...";
	next;
	if(select("Múmia? O que aconteceu?:Fingir que não sabe de nada.") == 2)
		close;
	mes "[Suspicious Cat]";
	mes "Você me pegou! Droga... eu não esperava que minha reputação como saqueador de tumbas fosse ser manchada assim...";
	next;
	mes "[Suspicious Cat]";
	mes "Mas qual é, elas são tão estranhas! São múmias! Apenas múmias, sabe? Elas não deveriam ser tão fortes!";
	next;
	if(select("Múmias fortes? Onde elas estão?:Encerrar a conversa.") == 2) {
		mes "[Suspicious Cat]";
		mes "O queee~ você não quer ouvir minha história? Tudo bem! É melhor não mencionar meu nome em nenhum outro lugar!";
		close;
	}
	mes "[Suspicious Cat]";
	mes "Haha, é um segredo.";
	mes "Descobri enquanto estava roubando... Não, quero dizer, apenas andando pela pirâmide.";
	next;
	mes "[Suspicious Cat]";
	mes "Quando vi uma múmia vindo em minha direção, eu pensei 'tanto faz' e não me importei. Mas então ele me derrubou com um tapa na cara! Como ele pôde fazer isso com um gato fofo como eu! Seu monstro enfaixado... você vai ver só!";
	next;
	if(select("Eu gostaria de ir também.:Encerrar a conversa.") == 2)
		close;
	mes "[Suspicious Cat]";
	mes "De jeito nenhum! Levei tanto tempo para encontrar aquele caminho!!";
	next;
	mes "[Suspicious Cat]";
	mes "Hmm... mas se você insiste, me dê 5.000 Zeny. Eu guiarei você.";
	mes "O quê~? Não me olhe assim! Você faz ideia de como é difícil para um gato sobreviver em Midgard!";
	next;
	mes "[Suspicious Cat]";
	mes "^FF0000Mas pensando bem, um fraco como você não sobreviverá lá. Talvez por um segundo, se tiver sorte?^000000";
	mes "Bem, não é da minha conta. Mas eu avisei, de qualquer forma!";
	next;
	switch(select("Pagar 5.000 Zeny e partir.:Esqueça.")) {
	case 1:
		if (Zeny < 5000) {
			mes "[Suspicious Cat]";
			mes "Aonde você pensa que vai! Você está muito quebrado!";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[Suspicious Cat]";
		mes "Bah, que papo é esse de dinheiro velho... você não tem nada brilhante ou reluzente?!";
		mes "Ahh, apenas me siga!";
		close2;
		warp "moc_prydn1",93,96;
		end;
	case 2:
		mes "[Suspicious Cat]";
		mes "Puxa, você nem tem 5.000 Zeny, né?";
		close;
	}
}
