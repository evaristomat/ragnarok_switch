PurchaseItem:

	.@item_id = getarg(0);
	.@req_setting = getarg(1);
	.@item_type = getarg(2);

	mes "[Logistics]";

	if (!.@item_type)
		mes "Você selecionou ^ff0000"+ getitemname(.@item_id) +"^000000.";
	else if (.@item_type == 1)
		mes "Existe apenas uma"+ callfunc("F_GetWeaponType",.@item_id) +": "+ getitemname(.@item_id) +".";
	else if (.@item_type == 2) {
		mes "Você selecionou '"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "A armadura para alguém com nível acima de 80 é: ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "Você selecionou 'Produção em Massa"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Existe apenas uma Produção em Massa"+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "Você selecionou 'Popularizada"+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "Existe apenas uma Popularizada"+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "Requer ^0000ff2.000^000000 Pontos KVM, e"+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";
	else if (.@req_setting == 4)
		mes "Requer ^0000ff1.200^000000 Pontos KVM, e"+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";
	else
		mes "Requer ^0000ff"+ .@prices[.@req_setting] +"^000000 Pontos KVM, e"+ .@name$ +", seus pontos são: ^580080"+ kvm_point +"^000000.";

	mes "Tem certeza de que deseja este item?";
	next;
	switch(select("Não, não vou comprar.:Sim, vou comprar.")) {
	case 1:
		mes "[Logistics]";
		mes "Você selecionou 'Não vou comprar'.";
		mes "Ao comprar um item, lembre-se de que não há reembolsos.";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[Logistics]";
			mes "Você comprou um"+ getitemname(.@item_id) +".";

			else
				mes "Seus Pontos de KVM foram reduzidos em ^0000ff"+ .@prices[.@req_setting] +"^000000 pontos, seus Pontos de KVM agora são ^580080"+kvm_point+"^000000.";

			mes "[Logistics]";
			if (!.@req_setting)

			else if (.@req_setting == 4)
