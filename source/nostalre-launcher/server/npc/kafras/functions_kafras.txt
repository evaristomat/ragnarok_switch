//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Funcionária Kafra]";
		mes "Bem-vindo ao";
		mes "Corporação Kafra.";
		mes "Os serviços Kafra";
		mes "estão sempre ao seu lado.";
		mes "Como posso ajudá-lo?";
		break;
	case 1:// Niflheim specific message
		mes "[Funcionária Kafra]";
		mes "^666666W-weeeelc-c-venha";
		mes "para o K-kaaafrrrra";
		mes "C-coorpoor-r-ratioooonn...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Funcionária Kafra]";
		mes "Bem-vindo. ^ff0000" + getguildname( getcharid(2) ) + "^000000 Membro.";
		mes "A Corporação Kafra estará com você onde quer que você vá.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Funcionária Kafra]";
		mes "Então, você veio de uma terra distante para estudar nossa cultura ou está apenas passeando?";
		mes "Em ambos os casos, por que não ficar um pouco?";
		mes "O ar está eternamente pesado com o";
		mes "aroma de agradáveis flores silvestres.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Funcionária Kafra]";
		mes "Com nossos muitos Kafra";
		mes "locais de serviço, você nunca está";
		mes "longe de casa.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Use Storage", "Use Guild Storage", "Rent a Pushcart", "Use Teleport Service", "Cancel";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Save","Use Storage","Cancel"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Use Storage","Cancel"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Save","Use Storage","Rent a Pushcart","Check Other Information","Cancel"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Use Storage","Rent a Pushcart","Check Other Information","Cancel"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Use Storage","Check Other Information","Cancel"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Save","Use Storage","Rent a Pushcart","Cancel"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Save","Use Storage","Check Other Information","Cancel"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Use Storage","Rent a Pushcart","Use Teleport Service","Check Other Information","Cancel"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Use Storage","Save","Rent a Pushcart","Cancel"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Save","Use Storage","Use Teleport Service","Rent a Pushcart","Check Other Information","Cancel"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Save")
			return;
		else if (.@K_Menu0$[.@j] == "Use Storage") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Use Teleport Service") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Funcionária Kafra]";
				mes "Por causa do ^FF0000Limitado";
				mes "Acordo de Transporte^000000, o";
				mes "A Kafra Corporation não pode";
				mes "fornecer serviços de teletransporte";
				mes "na República de Schwarzwald.";
				next;
				mes "[Funcionária Kafra]";
				mes "Pedimos que você por favor";
				mes "usar o serviço de dirigível";
				mes "em vez disso. Obrigado pela sua";
				mes "compreensão e cooperação.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Rent a Pushcart") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Check Other Information") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancel") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Use Guild Storage") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Funcionária Kafra]";
			mes "Sinto muito, mas outro membro da guilda está usando o armazenamento da guilda";
			mes "agora mesmo.  Por favor, espere até que essa pessoa termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Funcionária Kafra]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666S-s-ssoooo,";
			mes "você é a-a-aaaa";
			mes "Nooviiiice... N-precisa";
			mes "B-básico sssskill nível l 6...^000000";
			return;
		}
		mes "Me desculpe, mas você";
		mes "preciso do novato";
		mes "Nível de habilidade básica 6 a";
		mes "use o serviço de armazenamento.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Funcionária Kafra]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeny...";
					mes "M-mais z-zeny...!";
					mes "N-preciso 150... zeny...";
					mes "Ergh! To-tomando sangue~!^000000";
					return;
				}
				// Standard Message
				mes "Me desculpe, mas você não";
				mes "tenha zeny suficiente para usar";
				mes "o Serviço de Armazenamento. Nosso";
				mes "A taxa de acesso ao armazenamento é " + .@fee + " Zeny.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Funcionária Kafra]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Obrigado.. por... usar...^000000";
		}
		// Normal message
		else {
			mes "[Funcionária Kafra]";
			mes "Aqui, deixe-me abrir";
			mes "seu armazenamento para você.";
			mes "Obrigado por usar";
			mes "o Serviço Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Funcionária Kafra]";
	mes "Por favor escolha";
	mes "seu destino.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancel")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Funcionária Kafra]";
			mes "Sinto muito, mas você não tem";
			mes "zeny suficiente para o teleporte";
			mes "Serviço. A taxa para se teletransportar";
			mes "to " + @wrpD$[.@j] + " is " + @wrpP[.@j] + " Zeny.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		// Ragnarok Solo: the client data only has the renewal Izlude
		warp "izlude", 128, 98;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mjolnir Dead Pit") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Orc Dungeon") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Funcionária Kafra]";
		mes "Sinto muito, mas o";
		mes "Serviço de aluguel de carrinhos";
		if (checkre(0) == true)
			mes "está disponível apenas para classes Merchant.";
		else {
			mes "está disponível apenas para comerciantes,";
			mes "Ferreiros, Mestres Ferreiros,";
			mes "Alquimistas e Bioquímicos.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Funcionária Kafra]";
		mes "Você já tem";
		mes "um carrinho de mão equipado.";
		mes "Infelizmente, não podemos";
		mes "alugue mais de um para";
		mes "cada cliente por vez.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Funcionária Kafra]";
		mes "Você só pode alugar um carrinho depois de aprender o \"Carrinho de empurrar\" habilidade.";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Funcionária Kafra]";
		mes "O aluguel do carrinho";
		mes "a taxa é " + .@rental_fee + " Zeny. Iria";
		mes "você gosta de alugar um Pushcart?";
		next;
		if (select("Alugue um carrinho de mão.:Cancelar") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Funcionária Kafra]";
			mes "Me desculpe, mas você";
			mes "não tenho o suficiente";
			mes "zeny para pagar o carrinho";
			mes "taxa de aluguel de " + .@rental_fee + " Zeny.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

		mes "Obrigado pelo seu patrocínio.";

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Digite sua senha de armazenamento:";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Senha de armazenamento errada.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " orgulhosamente apresenta-lhe um novo serviço:";
		mes "Proteção adicional de armazenamento com senha.";
		next;
		if (select( "Definir nova senha -> 5000z", "Cancelar" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Seu armazenamento é protegido por senha. O que você faria agora?";
	next;
	switch( select( "Alterar senha antiga -> 5000z", "Remover senha de armazenamento -> 1000z", "Cancelar" ) ) {
	case 1:
		mes .@npc_name$;
		mes "Primeiramente, insira seu ^0000FFsenha antiga^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Senha errada. Você não pode definir uma nova senha.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Por favor, digite sua senha antes de removê-la.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "A senha não foi removida.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Você não tem Zeny suficiente.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "Você limpou com sucesso sua senha de armazenamento.";
			mes "Obrigado por usar " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Senha errada. Não devolveremos seu 1000z.";
			mes "Por favor, da próxima vez digite a senha correta.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Agora digite seu ^FF0000nova senha^000000 para proteger seu armazenamento de ladrões.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "A senha não foi alterada.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Você não tem Zeny suficiente.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Você protegeu seu armazenamento com uma senha secreta.";
	mes "Obrigado por usar " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Insira um número 1000~10000000:";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Tried to fit storage password.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Você não pode usar uma senha tão grande.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Você não deve usar uma senha tão curta.";
		return 0;
	}
	return .@code_;
}
