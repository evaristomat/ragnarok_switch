prt_in,243,168,4	script	Jovovich	91,{
	mes "[Cabeleireiro Jovovich]";
	mes "Bem-vindo~!";
	mes "Como posso ajudar você?";
	next;
	mes "[Cabeleireiro Jovovich]";
	if (Sex == SEX_MALE) mes "Ah, não! Seu cabelo está danificado. Parece que você precisa de um tratamento profissional. Venha sentar aqui, por favor. Venha.";
	else {
		if (rand(20) > 11) {
			mes "Eh!? Ah, meu Deus! Ah, não, não, não, não! Seu cabelo está tããão danificado! Não é bom deixar seu cabelo assim.";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "Você permitiria que eu cuidasse do seu cabelo? Por favor?";
		}
		else mes "Uau! Seu cabelo ficaria perfeito se fosse tingido~ Que tal tingir seu cabelo para mudar um pouco?";
	}
	next;
	while(1) {
		switch(select("Tingir Cabelo:Dicas e Informações:Cancelar")) {
		case 1:
			mes "[Cabeleireiro Jovovich]";
			mes "Sim, sim, ótima escolha~";
			mes "Bem, então esta é sua chance de uma transformação, sua chance de florescer em beleza! Escolha a cor que você gostaria.";
			next;
			while(1) {
				if (.@choose_success == 1) {
					mes "[Cabeleireiro Jovovich]";
					mes "O que você acha? Você queria uma cor diferente?";
					next;
					if (select("Sim:Não") == 1) {
						mes "[Cabeleireiro Jovovich]";
						mes "Ok! Escolha a cor que você gostaria.";
						next;
					}
					else  {
						mes "[Cabeleireiro Jovovich]";
						mes "Hmm, estou um pouco desapontada. Eu queria ter feito um trabalho melhor. Mas prometo que farei melhor da próxima vez. Por favor, volte sempre~";
						close;
					}
				}

				while(1) {
					switch(select("Vermelho, por favor.:Amarelo, por favor.:Violeta, por favor.:Laranja, por favor.:Verde, por favor.:Azul, por favor.:Branco, por favor.:Preto, por favor.:Na verdade, gosto do meu cabelo como está.")) {
					case 1: set .@headpalette,8; break;
					case 2: set .@headpalette,1; break;
					case 3: set .@headpalette,2; break;
					case 4: set .@headpalette,3; break;
					case 5: set .@headpalette,4; break;
					case 6: set .@headpalette,5; break;
					case 7: set .@headpalette,6; break;
					case 8: set .@headpalette,7; break;
					case 9:
						if (.@choose_success != 0) {
							mes "[Cabeleireiro Jovovich]";
							mes "Você deve gostar da cor do seu cabelo~";
							close;
						}
						mes "[Cabeleireiro Jovovich]";
						mes "Eehh~? Você não vai tingir o cabelo? Estou um pouco triste...";
						close;
					}

					if (.@headpalette == getlook(VAR_HEADPALETTE)) {
						mes "[Cabeleireiro Jovovich]";
						mes "Eh? Mas essa é a cor de cabelo que você já tem. Por favor, escolha uma cor diferente.";
						next;
					}
					else {
						switch(.@headpalette) {
						case 1: callsub S_NoDye,976,"yellow"; break;
						case 2: callsub S_NoDye,978,"violet"; break;
						case 3: callsub S_NoDye,980,"orange"; break;
						case 4: callsub S_NoDye,979,"green"; break;
						case 5: callsub S_NoDye,981,"blue"; break;
						case 6: callsub S_NoDye,982,"white"; break;
						case 7: callsub S_NoDye,983,"black"; break;
						case 8: callsub S_NoDye,975,"red"; break;
						}
						if (Zeny < 1000) {
							mes "[Cabeleireiro Jovovich]";
							mes "A taxa é de 1000 zenys. Você não tem o suficiente...?";
							close;
						}

						switch(.@headpalette) {
						case 1: delitem 976,1; break; //Lemon_Dyestuffs
						case 2: delitem 978,1; break; //Violet_Dyestuffs
						case 3: delitem 980,1; break; //Orange_Dyestuffs
						case 4: delitem 979,1; break; //Darkgreen_Dyestuffs
						case 5: delitem 981,1; break; //Cobaltblue_Dyestuffs
						case 6: delitem 982,1; break; //White_Dyestuffs
						case 7: delitem 983,1; break; //Black_Dyestuffs
						case 8: delitem 975,1; break; //Scarlet_Dyestuffs
						}
						set Zeny, Zeny-1000;
						setlook VAR_HEADPALETTE,.@headpalette;
						set .@choose_success,1;
						break;
					}

				}

			}

		case 2:
			mes "[Cabeleireiro Jovovich]";
			mes "Quando você estiver se sentindo para baixo, quando levar um fora, quando quiser impressionar alguém, ou até mesmo quando apenas quiser se destacar...";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "Para aquele lugar e momento especial, você não gostaria de um estilo de cabelo só seu? Desde que você tenha os corantes apropriados, eu farei seu cabelo ficar maravilhoso.";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "E tente não se preocupar muito com a taxa. Adquirir beleza é o mesmo que adquirir qualquer outra coisa. Hehe~";
			next;
			mes "[Cabeleireiro Jovovich]";
			mes "* Taxas e Informações *";
			mes "- 1 item de Corante da cor de cabelo que você deseja.";
			mes "- Taxa de 1000 zenys.";
			close;

		case 3:
			mes "[Cabeleireiro Jovovich]";
			mes "Homens ou mulheres...";
			mes "Todos têm o direito e a obrigação de serem bonitos.";
			close;
		}

	}

S_NoDye:
	if (countitem(getarg(0)) == 0) {
		mes "[Cabeleireiro Jovovich]";
		mes "Eh?! Mas você precisa do item '"+getitemname(getarg(0))+"' para tingir seu cabelo"+getarg(1)+"...";
		close;
	}
	return;
}
